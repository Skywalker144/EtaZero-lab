cmake_minimum_required(VERSION 3.18)
project(etazero LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
set(SCHEMA_HEADER ${CMAKE_CURRENT_BINARY_DIR}/generated/etazero/schema.h)
add_custom_command(OUTPUT ${SCHEMA_HEADER}
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../python/etazero/schema.py ${SCHEMA_HEADER}
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/../python/etazero/schema.py ${CMAKE_CURRENT_SOURCE_DIR}/../python/etazero/games.py ${CMAKE_CURRENT_SOURCE_DIR}/../python/etazero/outcomes.py VERBATIM)
# Shared source primitives are compiled once, including for the CUDA backend.
add_library(etazero_base
    core/global.cpp core/hash.cpp core/rand.cpp core/sha2.cpp core/md5.cpp
    core/bsearch.cpp core/test.cpp core/timer.cpp core/datetime.cpp core/fileutils.cpp
    core/config_parser.cpp core/logger.cpp core/makedir.cpp core/multithread.cpp ${SCHEMA_HEADER})
target_include_directories(etazero_base PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/generated
    PRIVATE external/filesystem-1.5.8/include)
target_link_libraries(etazero_base PUBLIC Threads::Threads ZLIB::ZLIB)
target_compile_definitions(etazero_base PUBLIC COMPILE_MAX_BOARD_LEN=25)
target_compile_options(etazero_base PRIVATE -ffunction-sections -fdata-sections)
target_link_options(etazero_base INTERFACE -Wl,--gc-sections)
add_library(etazero_go_rules game/go/board.cpp game/go/boardhistory.cpp game/go/rules.cpp
    game/go/graphhash.cpp game/go/sgf_handicap.cpp)
target_link_libraries(etazero_go_rules PUBLIC etazero_base)
add_library(etazero_go_features neuralnet/sgfmetadata.cpp neuralnet/nninputs.cpp search/localpattern.cpp)
target_link_libraries(etazero_go_features PUBLIC etazero_go_rules)
add_library(etazero_sgf dataio/sgf.cpp dataio/files.cpp)
target_link_libraries(etazero_sgf PUBLIC etazero_go_features)
target_include_directories(etazero_sgf PRIVATE external/filesystem-1.5.8/include)
target_compile_options(etazero_sgf PRIVATE -ffunction-sections -fdata-sections)
add_library(etazero_xiangqi_rules
    external/pikafish/bitboard.cpp external/pikafish/attacks.cpp external/pikafish/movegen.cpp
    external/pikafish/position.cpp external/pikafish/nnue/features/k6r2nc2.cpp)
set_property(TARGET etazero_xiangqi_rules PROPERTY CXX_STANDARD 20)
target_compile_definitions(etazero_xiangqi_rules PUBLIC IS_64BIT)
target_compile_options(etazero_xiangqi_rules PRIVATE -ffunction-sections -fdata-sections)
target_link_options(etazero_xiangqi_rules INTERFACE -Wl,--gc-sections)
add_library(etazero_game game/game.cpp game/gomoku.cpp game/hex.cpp game/rules.cpp
    game/go.cpp game/go_position.cpp game/xiangqi.cpp)
set_property(SOURCE game/xiangqi.cpp PROPERTY COMPILE_OPTIONS -std=c++20)
target_link_libraries(etazero_game PUBLIC etazero_sgf etazero_xiangqi_rules)
add_library(etazero_neuralnet neuralnet/batcher.cpp neuralnet/random_evaluator.cpp neuralnet/muzero/batcher.cpp)
target_link_libraries(etazero_neuralnet PUBLIC etazero_game)
add_library(etazero_search_common search/search_math.cpp search/searchresults.cpp search/searchparams.cpp
    search/gumbel.cpp search/subtree_value_bias.cpp search/timecontrols.cpp search/time_budget.cpp search/pattern_bonus.cpp)
target_link_libraries(etazero_search_common PUBLIC etazero_neuralnet)
add_library(etazero_search search/search.cpp search/gumbel_search.cpp search/analysis.cpp)
target_link_libraries(etazero_search PUBLIC etazero_search_common)
add_library(etazero_muzero search/muzero/search.cpp search/muzero/gumbel_search.cpp)
target_link_libraries(etazero_muzero PUBLIC etazero_search_common)
add_library(etazero_dataio dataio/record.cpp)
target_link_libraries(etazero_dataio PUBLIC etazero_search_common)
add_library(etazero_program program/opening.cpp program/search_limits.cpp program/sampling.cpp program/match.cpp program/setup.cpp)
target_link_libraries(etazero_program PUBLIC etazero_search etazero_muzero etazero_dataio)
foreach(target etazero_game etazero_neuralnet etazero_search_common etazero_search etazero_dataio etazero_program)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
add_executable(rules_replay tools/rules_replay.cpp)
target_include_directories(rules_replay PRIVATE external)
target_link_libraries(rules_replay PRIVATE etazero_game)
option(ETAZERO_WITH_TORCH "Build native model inference and commands" ON)
if(ETAZERO_WITH_TORCH)
    find_package(Torch REQUIRED)
    separate_arguments(TORCH_FLAGS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
    target_compile_options(etazero_base PUBLIC ${TORCH_FLAGS})
    target_compile_options(etazero_xiangqi_rules PUBLIC ${TORCH_FLAGS})
    add_library(etazero_torch neuralnet/torch_backend.cpp neuralnet/prepared_model.cpp neuralnet/backend_factory.cpp)
    target_include_directories(etazero_torch PRIVATE external)
    target_link_libraries(etazero_torch PUBLIC etazero_neuralnet "${TORCH_LIBRARIES}")
    set(SOURCE_CONFIG_HEADER ${CMAKE_CURRENT_BINARY_DIR}/generated/etazero/source_config.h)
    add_custom_command(OUTPUT ${SOURCE_CONFIG_HEADER}
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../python/etazero/source_config_header.py ${SOURCE_CONFIG_HEADER}
        DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/../python/etazero/source_config_header.py ${CMAKE_CURRENT_SOURCE_DIR}/../python/etazero/config_files.py ${CMAKE_CURRENT_SOURCE_DIR}/../python/etazero/config.py ${CMAKE_CURRENT_SOURCE_DIR}/../python/etazero/engine_config.py VERBATIM)
    add_executable(etazero command/main.cpp command/session.cpp command/source_analysis.cpp ${SOURCE_CONFIG_HEADER})
    target_link_libraries(etazero PRIVATE etazero_torch etazero_program)
    target_include_directories(etazero PRIVATE external)
    add_executable(inference_benchmark tools/inference_benchmark.cpp)
    target_link_libraries(inference_benchmark PRIVATE etazero_torch)
    # MuZero retains its own request and node layouts.
    add_library(etazero_muzero_torch neuralnet/muzero/torch_backend.cpp neuralnet/muzero/backend_factory.cpp)
    target_link_libraries(etazero_muzero_torch PUBLIC etazero_torch "${TORCH_LIBRARIES}")
    target_link_libraries(etazero PRIVATE etazero_muzero_torch)
endif()
include(cuda.cmake)
if(ETAZERO_WITH_TORCH AND ETAZERO_WITH_CUDA)
    target_compile_definitions(etazero_torch PRIVATE ETAZERO_WITH_CUDA)
    target_compile_definitions(etazero_muzero_torch PRIVATE ETAZERO_WITH_CUDA)
    target_link_libraries(etazero_muzero_torch PUBLIC etazero_cuda)
    target_link_libraries(etazero_torch PUBLIC etazero_cuda)
endif()
