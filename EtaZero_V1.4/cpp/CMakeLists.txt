cmake_minimum_required(VERSION 3.18)
project(etazero LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
set(SCHEMA_HEADER ${CMAKE_CURRENT_BINARY_DIR}/generated/etazero/schema.h)
add_custom_command(OUTPUT ${SCHEMA_HEADER}
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../python/etazero/schema.py ${SCHEMA_HEADER}
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/../python/etazero/schema.py VERBATIM)
add_library(etazero_core src/game/rules.cpp src/search/search.cpp src/search/search_math.cpp src/inference/batcher.cpp src/selfplay/record.cpp src/selfplay/opening.cpp src/selfplay/search_limits.cpp src/selfplay/sampling.cpp src/inference/random_evaluator.cpp ${SCHEMA_HEADER})
target_include_directories(etazero_core PUBLIC include ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(etazero_core PUBLIC Threads::Threads ZLIB::ZLIB OpenSSL::Crypto)
target_compile_options(etazero_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(etazero_muzero src/muzero/batcher.cpp src/muzero/search.cpp)
target_link_libraries(etazero_muzero PUBLIC etazero_core)
option(ETAZERO_WITH_TORCH "Build native model inference and commands" ON)
if(ETAZERO_WITH_TORCH)
    find_package(Torch REQUIRED)
    separate_arguments(TORCH_FLAGS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
    target_compile_options(etazero_core PUBLIC ${TORCH_FLAGS})
    add_library(etazero_torch src/inference/torch_backend.cpp src/inference/prepared_model.cpp)
    target_link_libraries(etazero_torch PUBLIC etazero_core "${TORCH_LIBRARIES}")
    add_executable(etazero src/commands/main.cpp)
    target_link_libraries(etazero PRIVATE etazero_torch)
    # MuZero retains its own request and node layouts.
    add_library(etazero_muzero_torch src/muzero/torch_backend.cpp)
    target_link_libraries(etazero_muzero_torch PUBLIC etazero_muzero etazero_torch "${TORCH_LIBRARIES}")
    target_link_libraries(etazero PRIVATE etazero_muzero_torch)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(muzero_search_test tests/muzero_search_test.cpp)
    target_link_libraries(muzero_search_test PRIVATE etazero_muzero)
    add_test(NAME muzero_search COMMAND muzero_search_test)
    set_tests_properties(muzero_search PROPERTIES TIMEOUT 30)
    if(ETAZERO_WITH_TORCH)
        add_executable(inference_probe tests/inference_probe.cpp)
        target_link_libraries(inference_probe PRIVATE etazero_torch)
        add_executable(muzero_inference_probe tests/muzero_inference_probe.cpp)
        target_link_libraries(muzero_inference_probe PRIVATE etazero_muzero_torch)
        add_executable(model_reload_probe tests/model_reload_probe.cpp)
        target_link_libraries(model_reload_probe PRIVATE etazero_muzero_torch)
    endif()
    add_executable(core_test tests/core_test.cpp)
    target_link_libraries(core_test PRIVATE etazero_core)
    add_test(NAME core COMMAND core_test)
    add_executable(opening_test tests/opening_test.cpp)
    target_link_libraries(opening_test PRIVATE etazero_core)
    add_test(NAME opening COMMAND opening_test)
    add_executable(search_features_test tests/search_features_test.cpp)
    target_link_libraries(search_features_test PRIVATE etazero_core)
    add_test(NAME search_features COMMAND search_features_test)
    add_executable(record_contract_test tests/record_contract_test.cpp)
    target_link_libraries(record_contract_test PRIVATE etazero_core)
    add_test(NAME record_contract COMMAND record_contract_test)
    add_executable(graph_test tests/graph_test.cpp)
    target_link_libraries(graph_test PRIVATE etazero_core)
    add_test(NAME graph COMMAND graph_test)
    add_executable(search_corrections_test tests/search_corrections_test.cpp)
    target_link_libraries(search_corrections_test PRIVATE etazero_core)
    add_test(NAME search_corrections COMMAND search_corrections_test)
    add_executable(sampling_test tests/sampling_test.cpp)
    target_link_libraries(sampling_test PRIVATE etazero_core)
    add_test(NAME sampling COMMAND sampling_test)
    set_tests_properties(core opening search_features record_contract graph search_corrections sampling PROPERTIES TIMEOUT 30)
endif()
